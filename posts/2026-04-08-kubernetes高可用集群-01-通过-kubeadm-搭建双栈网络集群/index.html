环境准备

  • Kubernetes 版本 v1.28

  • 集群架构设计

    • 控制平面 (Control Plane)：由 3 台 Master 节点组成。每台节点同时运行 kube-apiserver、kube-scheduler、kube-controller-manager 以及 etcd 存储服务
    • 负载均衡 (Load Balancer)：1 台独立的 LB 节点（这里使用HAProxy），负责将 kubectl 和 Worker 节点的流量分发给 3 台 Master 的 API Server
    • 工作节点 (Worker Nodes)：2 台节点负责运行实际的业务 Pod

  • 节点规划与网络配置

角色IPv4IPv6用途

  • 密钥登录

    • Ansible 控制机 -> 所有集群节点
    • 集群内部维度：Master -> Worker (非必须)

  • 核心 Playbook 文件 deploy_k8s.yml
    • 配置独立HAProxy LB 节点
    • 配置所有 K8s 节点 (Master & Worker)
      • 安装容器 containerd
      • 安装 kubeadm、kubelet 和 kubectl
      • 配置防火墙
      • 安装 Helm (仅 Master1)
---
- name: 0. 配置 LB 负载均衡器
  hosts: lb
  become: yes
  tasks:
    - name: 安装 HAProxy
      apt: name=haproxy state=present update_cache=yes

    - name: 配置 HAProxy 双栈转发 (监听 6443)
      blockinfile:
        path: /etc/haproxy/haproxy.cfg
        block: |
          frontend k8s-api
              bind *:6443
              bind :::6443
              mode tcp
              default_backend k8s-masters
          backend k8s-masters
              mode tcp
              balance roundrobin
              server m1 {{ hostvars['master1']['local_ipv4'] }}:6443 check
              server m2 {{ hostvars['master2']['local_ipv4'] }}:6443 check
              server m3 {{ hostvars['master3']['local_ipv4'] }}:6443 check
      notify: restart_haproxy

  handlers:
    - name: restart_haproxy
      systemd: name=haproxy state=restarted

- name: 1. 初始化所有 K8s 节点环境 (v1.28)
  hosts: k8s_all
  become: yes
  vars:
    k8s_version: "1.28"
    k8s_full_version: "1.28.10-1.1" 
  tasks:
    # --- 基础依赖安装 ---
    - name: 安装基础工具
      apt:
        name: [curl, gnupg, apt-transport-https, ca-certificates]
        state: present
        update_cache: yes

    - name: 设置主机名
      hostname: name="{{ inventory_hostname }}"

    - name: 同步 /etc/hosts
      blockinfile:
        path: /etc/hosts
        block: |
          10.0.0.170  haproxy
          10.0.0.181  master1
          10.0.0.182  master2
          10.0.0.183  master3
          10.0.0.191  worker1
          10.0.0.192  worker2

    - name: 配置防火墙 (简单直接方案)
      shell: |
        ufw allow from 10.0.0.0/24
        ufw allow from 6666:6666:2000:128::/64
        ufw --force enable

    - name: 彻底禁用 Swap
      shell: |
        swapoff -a
        sed -i '/swap/d' /etc/fstab

    # --- 内核模块与参数 ---
    - name: 内核配置 (模块与参数)
      copy:
        dest: "{{ item.path }}"
        content: "{{ item.content }}"
      with_items:
        - { path: '/etc/modules-load.d/k8s.conf', content: "overlay\nbr_netfilter\n" }
        - { path: '/etc/sysctl.d/k8s.conf', content: "net.bridge.bridge-nf-call-iptables=1\nnet.bridge.bridge-nf-call-ip6tables=1\nnet.ipv4.ip_forward=1\nnet.ipv6.conf.all.forwarding=1\n" }

    - name: 立即应用内核配置
      shell: |
        modprobe overlay
        modprobe br_netfilter
        sysctl --system

    # --- 容器运行时 (修正增强版) ---
    - name: 安装并配置 Containerd
      block:
        - name: 安装 Containerd 软件包
          apt: 
            name: containerd
            state: present

        - name: 生成并优化 Containerd 配置
          shell: |
            mkdir -p /etc/containerd
            containerd config default > /etc/containerd/config.toml
            # 1. 开启 SystemdCgroup (K8s 官方推荐)
            sed -i 's/SystemdCgroup = false/SystemdCgroup = true/g' /etc/containerd/config.toml
            # 2. 显式指定 Pause 镜像版本 (匹配 v1.28)
            sed -i 's|sandbox_image = ".*"|sandbox_image = "registry.k8s.io/pause:3.9"|g' /etc/containerd/config.toml
          
        - name: 启动并确保 Containerd 运行
          systemd:
            name: containerd
            state: restarted
            enabled: yes

    # --- K8s 工具链 ---
    - name: 添加 K8s 源
      shell: |
        mkdir -p -m 755 /etc/apt/keyrings
        curl -fsSL https://pkgs.k8s.io/core:/stable:/v{{ k8s_version }}/deb/Release.key | gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg
        echo "deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v{{ k8s_version }}/deb/ /" > /etc/apt/sources.list.d/kubernetes.list

    - name: 安装 K8s 工具
      apt:
        name: 
          - "kubelet={{ k8s_full_version }}"
          - "kubeadm={{ k8s_full_version }}"
          - "kubectl={{ k8s_full_version }}"
        update_cache: yes

    - name: 版本锁定
      shell: apt-mark hold kubelet kubeadm kubectl

    - name: Kubelet 绑定双栈 IP
      lineinfile:
        path: /etc/default/kubelet
        # 这里的 ansible_host 现在是 IPv6，所以我们用 local_ipv4 和 ansible_host 组合
        line: 'KUBELET_EXTRA_ARGS="--node-ip={{ local_ipv4 }},{{ ansible_host }}"'
        create: yes

- name: 2. 安装 Helm (Master1)
  hosts: master1
  become: yes
  tasks:
    - name: 一键安装 Helm
      shell: curl https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 | bash
  • 基础环境自动化部署流程
ansible-playbook -i k8s-hosts deploy_k8s.yml

部署节点与初始化

  • 在 Master1 初始化集群
kubeadm init \
  --control-plane-endpoint "10.0.0.170:6443" \
  --pod-network-cidr=10.244.0.0/16,fd00:10:244::/64 \
  --service-cidr=10.96.0.0/16,fd00:10:96::/112 \
  --upload-certs
  • 设置 kubectl Access
mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config

安装 Cilium CNI 插件

  • 在 Master1 部署
CILIUM_CLI_VERSION=$(curl -s https://raw.githubusercontent.com/cilium/cilium-cli/main/stable.txt)
CLI_ARCH=amd64
if [ "$(uname -m)" = "aarch64" ]; then CLI_ARCH=arm64; fi
curl -L --fail --remote-name-all https://github.com/cilium/cilium-cli/releases/download/${CILIUM_CLI_VERSION}/cilium-linux-${CLI_ARCH}.tar.gz{,.sha256sum}
sha256sum --check cilium-linux-${CLI_ARCH}.tar.gz.sha256sum
tar xzvf cilium-linux-${CLI_ARCH}.tar.gz
sudo mv cilium /usr/local/bin
rm cilium-linux-${CLI_ARCH}.tar.gz{,.sha256sum}
  • 一键部署 Cilium (适配双栈)
cilium install \
  --set ipv6.enabled=true \
  --set ipv4.enabled=true \
  --set ipam.mode=cluster-pool \
  --set ipam.operator.clusterPoolIPv4PodCIDRList="10.244.0.0/16" \
  --set ipam.operator.clusterPoolIPv4MaskSize=24 \
  --set ipam.operator.clusterPoolIPv6PodCIDRList="fd00:10:244::/64" \
  --set ipam.operator.clusterPoolIPv6MaskSize=80 \
  --set k8sServiceHost=10.0.0.170 \
  --set k8sServicePort=6443
  • 验证状态
cilium status --wait

负载均衡器 MetalLB

  • 确保 Kubernetes 的网络代理（kube-proxy）开启了 Strict ARP 模式，这是 MetalLB L2 模式正常工作的前提
# 将 strictARP 的值从 false 改为 true
# 将空字符串改为 "ipvs" , mode: "ipvs" 
kubectl edit configmap -n kube-system kube-proxy

# 重启 kube-proxy
kubectl rollout restart daemonset kube-proxy -n kube-system
  • 安装 MetalLB
# 应用安装文件
kubectl apply -f https://raw.githubusercontent.com/metallb/metallb/v0.14.3/config/manifests/metallb-native.yaml

# 验证状态
kubectl get pods -n metallb-system
  • 配置 MetalLB
    • 创建 IP 地址池 (双栈配置), 创建文件 metallb-config.yaml
apiVersion: metallb.io/v1beta1
kind: IPAddressPool
metadata:
  name: dual-stack-pool
  namespace: metallb-system
spec:
  addresses:
  - 10.0.0.200-10.0.0.250              # 替换为你自己的 IPv4 范围
  - 6666:6666:2000:128::200-6666:6666:2000:128::250 # 替换为你的专用 IPv6 范围
---
apiVersion: metallb.io/v1beta1
kind: L2Advertisement
metadata:
  name: l2-adv
  namespace: metallb-system
spec:
  ipAddressPools:
  - dual-stack-pool
  • 应用配置
kubectl apply -f metallb-config.yaml
  • 查看地址池列表
kubectl get ipaddresspool -n metallb-system
# 或
kubectl get ipaddresspool -A

常见问题与修复方法

  • Node 加入后仍留在 NotReady 中
journalctl -u kubelet -f

# Common cause: CNI plugin not installed, or containerd not configured correctly
# Fix: verify SystemdCgroup = true in /etc/containerd/config.toml
grep SystemdCgroup /etc/containerd/config.toml
sudo systemctl restart containerd kubelet
  • kubeadm init 失败，显示“端口已在使用中”
# A previous failed init left state behind
sudo kubeadm reset -f
sudo rm -rf /etc/cni/net.d
sudo rm -rf $HOME/.kube
# Then re-run kubeadm init
  • Pods卡在 ContainerCreating
kubectl describe pod <pod-name>
# Look for: "network: failed to set bridge addr"
# Usually means CNI wasn't installed or is misconfigured

# Check Calico pods are all Running
kubectl get pods -n kube-system | grep calico
  • 无法从 kubectl 访问 API 服务器
# Check the API server is running
sudo systemctl status kubelet
kubectl get pods -n kube-system | grep apiserver

# Check the port is open in your Security Group (6443)
# Check the server address in ~/.kube/config matches public IP
  • 忘了用连接命令
# On the control plane — regenerate it
kubeadm token create --print-join-command
  • Cilium 报错 node CIDR size is too big level=fatal msg=“Unable to init cluster-pool allocator” module=operator.operator-controlplane.leader-lifecycle.legacy-cell error=“unable to initialize IPv6 allocator: New CIDR set failed; the node CIDR size is too big”
# 观察 Operator 日志
kubectl logs -n kube-system deployment/cilium-operator
  • crictl ps 报错 WARN[0000] runtime connect using default endpoints: [unix:///var/run/dockershim.sock unix:///run/containerd/containerd.sock unix:///run/crio/crio.sock unix:///var/run/cri-dockerd.sock]. As the default settings are now deprecated, you should set the endpoint instead. ERRO[0000] validate service connection: validate CRI v1 runtime API for endpoint “unix:///var/run/dockershim.sock”: rpc error: code = Unavailable desc = connection error: desc = “transport: Error while dialing: dial unix /var/run/dockershim.sock: connect: no such file or directory” WARN[0000] image connect using default endpoints: [unix:///var/run/dockershim.sock unix:///run/containerd/containerd.sock unix:///run/crio/crio.sock unix:///var/run/cri-dockerd.sock]. As the default settings are now deprecated, you should set the endpoint instead. ERRO[0000] validate service connection: validate CRI v1 image API for endpoint “unix:///var/run/dockershim.sock”: rpc error: code = Unavailable desc = connection error: desc = “transport: Error while dialing: dial unix /var/run/dockershim.sock: connect: no such file or directory”
# 生成配置文件
cat <<EOF | sudo tee /etc/crictl.yaml
