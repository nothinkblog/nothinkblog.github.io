Proxmox VE Template VM镜像制作

  • 在 PVE 宿主机制作 Debian 12 模板
#!/bin/bash

# 1. 检查并安装工具
if ! command -v virt-customize >/dev/null 2>&1; then
    apt-get update && apt-get install -y libguestfs-tools
fi

# === 变量定义 ===
VM_ID=1200
VM_NAME="debian12-template"
STORAGE="local-lvm"
BRIDGE="vmbr1"
IMAGE_URL="https://cloud.debian.org/images/cloud/bookworm/latest/debian-12-generic-amd64.qcow2"
IMAGE_NAME="debian-12-generic-amd64.qcow2"
USER_SSH_KEY=$(cat ~/.ssh/id_rsa.pub 2>/dev/null || echo "")

# 2. 处理镜像
cd /root
[ -f "$IMAGE_NAME" ] || wget "$IMAGE_URL"

virt-customize -a "$IMAGE_NAME" \
    --install qemu-guest-agent,cloud-init,locales \
    --run-command "sed -i 's/^# en_US.UTF-8 UTF-8/en_US.UTF-8 UTF-8/' /etc/locale.gen && locale-gen" \
    --run-command "systemctl enable qemu-guest-agent" \
    --truncate /etc/machine-id \
    --timezone "Asia/Shanghai"

# 3. 创建虚拟机 (如果已存在模板，需要手动删除后再跑)
echo "--- 准备创建虚拟机 ---"
qm destroy $VM_ID 2>/dev/null || true
qm create $VM_ID --name "$VM_NAME" --memory 2048 --cores 2 --cpu x86-64-v2 --net0 virtio,bridge=$BRIDGE --ostype l26

# 4. 导入磁盘
echo "--- 正在导入磁盘 ---"
qm importdisk $VM_ID "$IMAGE_NAME" $STORAGE

# 5. 【核心修复】精确抓取磁盘路径并剔除空格
# 逻辑：查找含有 unused 的行，提取冒号后的内容，并用 xargs 去除前后空格
DISK_PATH=$(qm config $VM_ID | grep 'unused' | awk -F': ' '{print $2}' | cut -d',' -f1 | xargs)

if [[ -z "$DISK_PATH" ]]; then
    echo "错误：无法获取磁盘路径，尝试回退到默认命名..."
    DISK_PATH="$STORAGE:vm-$VM_ID-disk-0"
fi

echo "确认磁盘路径为: [$DISK_PATH]"

# 6. 配置硬件
# 使用双引号包裹变量，防止路径中存在不可见字符
qm set $VM_ID --scsihw virtio-scsi-pci --scsi0 "$DISK_PATH",discard=on,ssd=1
qm set $VM_ID --delete unused0
qm set $VM_ID --ide2 $STORAGE:cloudinit
qm set $VM_ID --boot order=scsi0 --agent enabled=1
qm set $VM_ID --serial0 socket --vga serial0
qm set $VM_ID --ipconfig0 ip=dhcp --citype nocloud

if [ -n "$USER_SSH_KEY" ]; then
    qm set $VM_ID --sshkeys <(echo "$USER_SSH_KEY")
fi

# 7. 转换模板
echo "正在转换为模板..."
qm template $VM_ID
echo "✅ 完成！"

#!/bin/bash

# 遇到错误立即停止
set -e

# === 变量定义 (固定 ID 1300) ===
VM_ID=1300
VM_NAME="debian13-template"
STORAGE="local-lvm"
BRIDGE="vmbr1"
# 注意：Debian 13 目前是 Trixie (Testing 版)
IMAGE_URL="https://cloud.debian.org/images/cloud/trixie/latest/debian-13-generic-amd64.qcow2"
IMAGE_NAME="debian-13-trixie-generic-amd64.qcow2"
USER_SSH_KEY=$(cat ~/.ssh/id_rsa.pub 2>/dev/null || echo "")

# 1. 检查工具
if ! command -v virt-customize >/dev/null 2>&1; then
    echo "正在安装 libguestfs-tools..."
    apt-get update && apt-get install -y libguestfs-tools
fi

# 2. 下载镜像
cd /root
if [ ! -f "$IMAGE_NAME" ]; then
    echo "正在下载 Debian 13 镜像..."
    wget -O "$IMAGE_NAME" "$IMAGE_URL"
fi

# 3. 预处理镜像
echo "--- 正在预处理镜像 (注入 Agent & 清理) ---"
virt-customize -a "$IMAGE_NAME" \
    --install qemu-guest-agent,cloud-init,locales \
    --run-command "sed -i 's/^# en_US.UTF-8 UTF-8/en_US.UTF-8 UTF-8/' /etc/locale.gen && locale-gen" \
    --run-command "systemctl enable qemu-guest-agent" \
    --truncate /etc/machine-id \
    --timezone "Asia/Shanghai"

# 4. 安全清理旧的 1300
echo "--- 检查并清理旧的 VM/模板 $VM_ID ---"
if qm status $VM_ID >/dev/null 2>&1; then
    if qm config $VM_ID | grep -q "template: 1"; then
        echo "检测到 ID $VM_ID 是模板，正在解除锁定..."
        sed -i '/template: 1/d' /etc/pve/qemu-server/$VM_ID.conf
    fi
    echo "正在删除旧的 ID $VM_ID..."
    qm destroy $VM_ID --purge
fi

# 5. 创建虚拟机外壳
echo "--- 创建虚拟机外壳 (CPU: x86-64-v2) ---"
# 修正点：CPU 类型去掉 -aes 后缀
qm create $VM_ID --name "$VM_NAME" --memory 2048 --cores 2 --cpu x86-64-v2 --net0 virtio,bridge=$BRIDGE --ostype l26

# 6. 导入磁盘
echo "--- 导入磁盘 ---"
qm importdisk $VM_ID "$IMAGE_NAME" $STORAGE

# 7. 精确绑定磁盘 (修复空格问题)
DISK_PATH=$(qm config $VM_ID | grep 'unused' | awk -F': ' '{print $2}' | cut -d',' -f1 | xargs)

if [ -z "$DISK_PATH" ]; then
    echo "无法自动获取磁盘路径，使用默认路径猜测..."
    DISK_PATH="$STORAGE:vm-$VM_ID-disk-0"
fi

echo "确认磁盘路径: [$DISK_PATH]"

# 8. 配置硬件与 Cloud-Init
qm set $VM_ID --scsihw virtio-scsi-pci --scsi0 "$DISK_PATH",discard=on,ssd=1
qm set $VM_ID --delete unused0
qm set $VM_ID --ide2 $STORAGE:cloudinit
qm set $VM_ID --boot order=scsi0 --agent enabled=1,fstrim_cloned_disks=1
qm set $VM_ID --serial0 socket --vga serial0
qm set $VM_ID --ipconfig0 ip=dhcp --citype nocloud

# 注入 SSH Key
if [ -n "$USER_SSH_KEY" ]; then
    echo "注入 SSH 公钥..."
    qm set $VM_ID --sshkeys <(echo "$USER_SSH_KEY")
fi

# 9. 转换模板
echo "--- 正在转换为模板 ---"
qm template $VM_ID

echo "------------------------------------------------"
echo "✅ Debian 13 (Trixie) 模板制作完成！"
echo "VM ID: $VM_ID"
echo "------------------------------------------------"

在 PVE 宿主机制作 Ubuntu 24.04 模板

#!/bin/bash

# =================================================================
# 脚本名称: build-ubuntu2404-pve-template.sh
# 适用环境: Proxmox VE 8.x/9.x | Ubuntu 24.04 LTS (Noble)
# =================================================================

set -e

# === 1. 变量定义 ===
VM_ID=2404
VM_NAME="ubuntu2404-template"
STORAGE="local-lvm"
BRIDGE="vmbr1"
IMAGE_URL="https://cloud-images.ubuntu.com/noble/current/noble-server-cloudimg-amd64.img"
IMAGE_NAME="ubuntu-24.04-server.img"
DISK_SIZE="20G"

# === 2. 环境检查 ===
if ! command -v virt-customize >/dev/null 2>&1; then
    echo "正在安装必备工具 libguestfs-tools..."
    apt-get update && apt-get install -y libguestfs-tools
fi

# === 3. 下载与镜像预处理 ===
echo "--- 正在处理 Ubuntu 24.04 镜像 ---"
cd /root
[ -f "$IMAGE_NAME" ] || wget -O "$IMAGE_NAME" "$IMAGE_URL"

echo "--- 正在将镜像扩容至 $DISK_SIZE ---"
qemu-img resize "$IMAGE_NAME" "$DISK_SIZE"

# 使用 virt-customize 注入配置
echo "--- 正在注入系统配置 (Agent, Cloud-Init) ---"
virt-customize -a "$IMAGE_NAME" \
    --install qemu-guest-agent,cloud-init,nfs-common \
    --timezone "Asia/Shanghai" \
    --run-command "truncate -s 0 /etc/machine-id" \
    --run-command "apt-get autoremove -y && apt-get clean"

# === 4. 【核心修复】安全清理旧的 VM/模板 ===
echo "--- 检查并清理旧的 VM/模板 $VM_ID ---"
if qm status $VM_ID >/dev/null 2>&1; then
    if qm config $VM_ID | grep -q "template: 1"; then
        echo "检测到 ID $VM_ID 是模板，正在解除锁定并清理..."
        # 强制删除配置文件中的模板标记以允许删除
        sed -i '/template: 1/d' /etc/pve/qemu-server/$VM_ID.conf
    fi
    qm destroy $VM_ID --purge
fi

# === 5. 创建虚拟机外壳 ===
echo "--- 创建虚拟机外壳 (CPU: x86-64-v2) ---"
# 修正点：CPU 类型去掉 -aes 后缀以适配 PVE 标准
qm create $VM_ID --name "$VM_NAME" \
    --memory 2048 --cores 2 --cpu x86-64-v2 \
    --net0 virtio,bridge=$BRIDGE \
    --ostype l26 --agent enabled=1,fstrim_cloned_disks=1

# === 6. 导入磁盘并精确绑定 ===
echo "--- 正在导入磁盘并配置属性 ---"
qm importdisk $VM_ID "$IMAGE_NAME" $STORAGE --format qcow2

# 【核心修复】自动获取路径并使用 xargs 彻底清除空格
DISK_PATH=$(qm config $VM_ID | grep 'unused' | awk -F': ' '{print $2}' | cut -d',' -f1 | xargs)

if [ -z "$DISK_PATH" ]; then
    echo "无法自动获取磁盘路径，使用默认路径猜测..."
    DISK_PATH="$STORAGE:vm-$VM_ID-disk-0"
fi

echo "确认磁盘路径: [$DISK_PATH]"

# 绑定磁盘并开启高级特性 (SSD/Discard)
qm set $VM_ID --scsihw virtio-scsi-pci --scsi0 "$DISK_PATH",discard=on,ssd=1
qm set $VM_ID --delete unused0

# === 7. Cloud-Init 与引导设置 ===
echo "--- 配置引导项与 Cloud-Init ---"
qm set $VM_ID --ide2 $STORAGE:cloudinit
qm set $VM_ID --boot order=scsi0
qm set $VM_ID --serial0 socket --vga serial0
qm set $VM_ID --ipconfig0 ip=dhcp
qm set $VM_ID --citype nocloud

# 注入本地 SSH 公钥
if [ -f ~/.ssh/id_rsa.pub ]; then
    echo "正在注入 SSH 公钥..."
    qm set $VM_ID --sshkeys ~/.ssh/id_rsa.pub
fi

# === 8. 转换模板 ===
echo "--- 正在转换为模板 ---"
qm template $VM_ID

echo "------------------------------------------------"
echo "✅ Ubuntu 24.04 模板 (ID: $VM_ID) 制作完成！"
echo "------------------------------------------------"

resource "proxmox_virtual_environment_vm" "vm_nodes" {
  for_each = local.vms

  node_name = "pve"
  vm_id     = each.value.vmid
  name      = each.value.hostname
  tags      = [each.key]

  # 克隆配置
  clone {
    vm_id = 1200 # 建议使用模板的 ID，如果是名称则需确保唯一
    full  = true
  }

  # 1. 硬件抽象层配置
  agent { enabled = true } # 强烈建议开启 QEMU Agent
  
  cpu {
    cores = each.value.cpu
    type  = "host" # 使用宿主机 CPU 特性，性能更好
  }

  memory {
    dedicated = each.value.memory
  }

  # 2. 磁盘配置 (VM 需要额外的 EFI 磁盘和 Cloud-init 驱动)
  disk {
    datastore_id = "local-lvm"
    interface    = "scsi0"
    size         = each.value.disk_size
  }

  # 修正 3: VM 必须指定一个控制启动顺序的选项，否则克隆后可能无法自动启动
  operating_system {
    type = "l26" # Linux 2.6+ 内核
  }

  # 3. 网络配置
  network_device {
    bridge = "vmbr1"
  }

  # 4. Cloud-Init 核心配置 (替代了 CT 的 initialization)
  initialization {
    datastore_id = "local-lvm"
    
    ip_config {
      ipv4 {
        address = each.value.ipv4
        gateway = "10.0.0.1"
      }
      ipv6 {
        address = each.value.ipv6
        gateway = "6666:6666:6666:128::eeee"
      }
    }

    user_account {
      username = "root"
      keys     = [trimspace(file("${path.module}/test.pub"))]
    }
    
    dns {
      servers = ["1.1.1.1", "8.8.8.8", "2606:4700:4700::1111", "2001:4860:4860::8888"]
    }
  }

  # 5. 自动执行 Ansible/脚本
  provisioner "remote-exec" {
    inline = [
      "until [ -f /var/lib/cloud/instance/boot-finished ]; do echo 'Waiting for cloud-init...'; sleep 1; done",
      "apt update && apt install -y locales",
      "sed -i '/en_US.UTF-8 UTF-8/s/^# //g' /etc/locale.gen",
      "locale-gen",
      "update-locale LC_ALL=en_US.UTF-8 LANG=en_US.UTF-8",
      "sed -i 's/#PasswordAuthentication yes/PasswordAuthentication no/' /etc/ssh/sshd_config",
      "sed -i 's/PasswordAuthentication yes/PasswordAuthentication no/' /etc/ssh/sshd_config",
      "echo 'PermitRootLogin prohibit-password' >> /etc/ssh/sshd_config",
      "systemctl restart ssh"
    ]

    connection {
      type        = "ssh"
      user        = "root"
      host        = split("/", each.value.ipv6)[0]
      private_key = file("${path.module}/test.key")
      timeout     = "5m"
    }
  }
}

Proxmox VE Ansible 应用与系统层

  • 创建好 VM或CT后交给 ansible 去初始化系统
# 初始化配置文件 system_init.yml
---
- name: VPS 系统环境初始化与监控配置
  hosts: all
  become: yes
  vars:
    # 基础软件包列表
    base_packages:
      - net-tools
      - htop
      - vim
      - tree
      - curl
      - prometheus-node-exporter

  tasks:
    - name: 1.1 更新 apt 缓存并升级系统
      apt:
        update_cache: yes
        upgrade: dist
        autoremove: yes

    - name: 1.2 安装基础软件包
      apt:
        name: "{{ base_packages }}"
        state: present

    - name: 2. 配置 UFW 防火墙
      block:
        - name: 确保安装 ufw
          apt:
            name: ufw
            state: present

        - name: 允许默认 SSH 端口 (22)
          ufw:
            rule: allow
            port: "22"
            proto: tcp

        - name: 允许特定来源访问 Node Exporter 端口 (9100)
          ufw:
            rule: allow
            from_ip: "6666:6622:6666:5faa:0:ce4e:9de8:65f7"
            port: "9100"
            proto: tcp
            comment: "grafana"

        - name: 启用 UFW 并设置默认拒绝入站
          ufw:
            state: enabled
            policy: deny
            direction: incoming

    - name: 3.1 消除 cloud-init 的 SSH 配置干扰
      file:
        path: /etc/ssh/sshd_config.d/50-cloud-init.conf
        state: absent  # 直接删除该文件，让主配置文件生效

    - name: 3.2 SSH 安全加固
      lineinfile:
        path: /etc/ssh/sshd_config
        regexp: "{{ item.regexp }}"
        line: "{{ item.line }}"
      loop:
        - { regexp: '^#?PasswordAuthentication', line: 'PasswordAuthentication no' }
        - { regexp: '^#?PermitRootLogin', line: 'PermitRootLogin prohibit-password' }
      notify: 重启 ssh 服务

  handlers:
    - name: 重启 ssh 服务
      service:
        name: ssh
        state: restarted

# 执行下面命令即可 
ansible-playbook -i hosts system_init.yml --limit test

Proxmox VE 启用 IPSet 防止用户更改IP

  • 启用 Proxmox 防火墙的 IP Set 绑定

    1. 在 Proxmox 宿主机层面开启限制
      • 数据中心级别：Firewall -> Enabled: Yes
      • 节点级别：Firewall -> Enabled: Yes
      • 容器 (VM 201) 级别：
        • 开启 IP 过滤（改 IP 即断网的核心）
          • Firewall -> Options -> IPfilter: Yes （IP 筛选参数）
        • 关闭 DHCP（防止用户乱搞）
          • Firewall -> Options -> DHCP: No (防止用户自建 DHCP 污染)
        • 网卡也需要开启防火墙
          • 使用命令查看：pct config 201 | grep net0 （显示 firewall=1 即可）

  • 容器级别也可以使用加固脚本（secure_ct.sh）

#!/bin/bash
VMID=$1
# 开启 IP 过滤（改 IP 即断网的核心）
pvesh set /nodes/pve/lxc/$VMID/firewall/options --ipfilter 1
# 关闭 DHCP（防止用户乱搞）
pvesh set /nodes/pve/lxc/$VMID/firewall/options --dhcp 0
  • 防止 MAC 地址欺骗
    1. 在 Terraform 的 network 部分显式指定 hwaddr 以及 firewall 参数
network {
    name   = "eth0"
    bridge = "vmbr1"
    
    # 1. 手动固定 MAC (建议前三个字节跟随 PVE 默认（bc:24:11），后三个字节自定义)
    # # 1. 固定 MAC 地址（防止用户伪造 MAC）
    hwaddr = "bc:24:11:aa:bb:01"
    
    # 2. 开启接口防火墙
    # 2. 告诉 PVE 此网卡启用防火墙逻辑
    firewall = true

    ip     = "10.0.0.101/24"
    gw     = "10.0.0.1"
    ip6    = "6666:6666:2000:128::201/64"
    gw6    = "6666:6666:2000:128::eeee"
  }
